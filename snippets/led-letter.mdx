## LedLetter circuit

Draw any capital letter `A`–`Z` with a 5×7 grid of 0402 SMD LEDs. Every lit pixel
gets its own `<led />` plus a current-limiting `<resistor />`, wired between
`power` and `gnd` and routed with `<trace />`.

### Alphabet data

The letter shapes are stored as a 5×7 bitmap (`1` = lit pixel) so placement is
entirely math-driven:

### LedLetter component

```tsx title="lib/LedLetter.tsx"
import { LedLetter } from "lib/LedLetter"

export const MyExample = () => <LedLetter letter="A" power="net.power" gnd="net.gnd" />
```

```tsx title="lib/LedLetter.tsx"
import { ALPHABET } from "./alphabet"

type LedLetterProps = {
  letter: string
  power: string
  gnd: string
}

const PITCH = 1.5
const RESISTANCE = "330"

export const LedLetter = ({ letter, power, gnd }: LedLetterProps) => {
  const grid = ALPHABET[letter.toUpperCase()] ?? ALPHABET.A
  return (
    <group>
      {grid.map((row, r) =>
        row.map((lit, c) => {
          if (!lit) return null
          const led = `LED${r}${c}`
          const res = `R${r}${c}`
          const x = c * PITCH
          const y = -r * PITCH
          return (
            <group key={led}>
              <led name={led} pos={power} neg={gnd} footprint="0402" x={x} y={y} />
              <resistor
                name={res}
                resistance={RESISTANCE}
                footprint="0402"
                left={power}
                x={x}
                y={y}
              />
              <trace from={`.${res} > .pin2`} to={`.${led} > .pos`} />
            </group>
          )
        })
      )}
    </group>
  )
}
```
